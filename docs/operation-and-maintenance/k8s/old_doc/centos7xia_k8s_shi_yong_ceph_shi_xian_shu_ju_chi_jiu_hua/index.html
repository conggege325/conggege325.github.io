          以下示例以三台centos7服务器为例：

          主机名 IP 角色

          node1 192.168.128.201 cephadm，ceph-mon，ceph-mgr，ceph-osd，ceph-mds，rgw

          node2 192.168.128.202 ceph-mon，ceph-mgr，ceph-osd，ceph-mds，rgw

          node3 192.168.128.203 ceph-mon，ceph-mgr，ceph-osd，ceph-mds，rgw

          设置主机名与时区（三台机器）：

           1# 设置时区
 2timedatectl set-timezone Asia/Shanghai
 3
 4# node1 执行
 5hostnamectl set-hostname node1
 6
 7# node2 执行
 8hostnamectl set-hostname node2
 9
10# node3 执行
11hostnamectl set-hostname node3

          网络设置（三台机器）：

          设置静态ip (可选设置，但是建议设置，这样避免路由器分配ip的不确定)：

           1vim /etc/sysconfig/network-scripts/ifcfg-ens33
 2# 删除：
 3BOOTPROTO=dhcp
 4# 添加：
 5BOOTPROTO=static
 6# 根据实际 ip 设置，例如：node1 为 192.168.128.201
 7
 8IPADDR=192.168.128.201
 9NETMASK=255.255.255.0
10GATEWAY=192.168.128.2
11DNS1=114.114.114.114
12DNS2=8.8.8.8

          设置hosts：

          1cat >> /etc/hosts <<EOF
2192.168.128.201 node1
3192.168.128.202 node2
4192.168.128.203 node3
5EOF

          关闭防火墙、seLinux（三台机器）：

          1sed -i 's/SELINUX=enforcing/SELINUX=disabled/g' /etc/selinux/config
2
3systemctl disable firewalld

          修改ssh配置（三台机器）：

          1vim /etc/ssh/sshd_config
2# 修改如下内容：
3PermitRootLogin yes
4PasswordAuthentication yes

          重启三台服务器，使上面的设置生效

          安装 docker （三台机器）：

          cephadm基于容器运行所有ceph组件，所有节点需要安装docker

           1# 查看系统版本
 2[root@node1 ~]# cat /etc/redhat-release
 3CentOS Linux release 7.5.1804 (Core)
 4
 5# 配置centos7阿里云yum源
 6[root@node1 ~]# curl -o /etc/yum.repos.d/CentOS-Base.repo https://mirrors.aliyun.com/repo/Centos-7.repo
 7
 8# 安装 python3 才能执行 cephadm 脚本
 9[root@node1 ~]# yum -y install python3 yum-utils
10
11# 安装docker
12[root@node1 ~]# yum-config-manager --add-repo http://mirrors.aliyun.com/docker-ce/linux/centos/docker-ce.repo
13[root@node1 ~]# yum install -y docker-ce docker-ce-cli containerd.io
14
15# docker配置国内源
16[root@node1 ~]# vim /etc/docker/daemon.json
17# 内容如下：
18{
19  "registry-mirrors":["https://docker.mirrors.ustc.edu.cn","https://registry.docker-cn.com","http://hub-mirror.c.163.com"],
20  "live-restore":true,
21  "exec-opts": ["native.cgroupdriver=systemd"]
22}
23
24# 启动 docker
25[root@node1 ~]# systemctl enable --now docker

          初始化集群（node1 执行）

            1# 下载安装脚本
  2[root@node1 ~]# curl -o ./cephadm https://github.com/ceph/ceph/raw/octopus/src/cephadm/cephadm
  3[root@node1 ~]# chmod +x cephadm
  4
  5# 此命令将生成 ceph yum 源
  6[root@node1 ~]# ./cephadm add-repo --release octopus
  7
  8# 备份 ceph yum 源并将其替换使用 阿里云 yum 源
  9[root@node1 ~]# cp /etc/yum.repos.d/ceph.repo{,.back}
 10[root@node1 ~]# sed -i 's#download.ceph.com#mirrors.aliyun.com/ceph#' /etc/yum.repos.d/ceph.repo
 11[root@node1 ~]# yum list | grep ceph
 12
 13# 安装 cephadm 到当前服务器(其实就是将 cephadm 复制到环境变量)

          部署MDS（node1上操作）

          使用 CephFS 文件系统需要一个或多个 MDS 守护程序。

          如果使用新的ceph fs卷接口来创建新文件系统，则会自动创建这些文件 部署元数据服务器：

          ceph orch apply mds –placement=" [ …]"

          CephFS 需要两个 Pools，cephfs-data 和 cephfs-metadata,分别存储文件数据和文件元数据

           1[root@node1 ~]# ceph osd pool create cephfs_data 64 64
 2pool 'cephfs_data' created
 3
 4[root@node1 ~]# ceph osd pool create cephfs_metadata 64 64
 5pool 'cephfs_metadata' created
 6
 7# 创建一个 CephFS, 名字为 cephfs
 8[root@node1 ~]# ceph fs new cephfs cephfs_metadata cephfs_data
 9new fs with metadata pool 3 and data pool 2
10
11[root@node1 ~]# ceph fs ls
12name: cephfs, metadata pool: cephfs_metadata, data pools: [cephfs_data ]
13
14[root@node1 ~]# ceph orch apply mds cephfs --placement="3 node1 node2 node3"

          部署RGW（node1上操作）

          Cephadm将radosgw部署为管理特定领域和区域的守护程序的集合，

          RGW是Ceph对象存储网关服务RADOS Gateway的简称，是一套基于LIBRADOS接口封装而实现的FastCGI服务，对外提供RESTful风格的对象存储数据访问和管理接口。

          使用 cephadm 时，radosgw 守护程序是通过mon配置数据库而不是通过ceph.conf 或命令行配置的。

          如果该配置尚未就绪，则 radosgw 守护进程将使用默认设置启动（默认绑定到端口 80）。

          在node1、node2和node3上部署3个服务于myorg领域和us-east-1区域的rgw守护进程（如果领域、区域不存在，会自动创建）：

          1[root@node1 ~]# ceph orch apply rgw myorg cn-east-1 --placement="3 node1 node2 node3"
2Scheduled rgw.myorg.cn-east-1 update...

          领域、区域组和区域也可以手动创建：

          1# 创建领域
2radosgw-admin realm create --rgw-realm=myorg --default
3# 创建区域组：
4radosgw-admin zonegroup create --rgw-zonegroup=default --master --default
5# 创建一个区域：
6radosgw-admin zone create --rgw-zonegroup=default --rgw-zone=cn-east-1 --master --default
