          K8S 的 StorageClass 使用 GlusterFS 实现数据持久化

          1[root@k8s-node1 ~]# mkdir -p heketi
2[root@k8s-node1 ~]# cd heketi/

          （1）创建storageclass服务配置（heketi-sc.yaml）

          heketi 认证的密码是base64转码后的，命令如下：

          1[root@k8s-node1 heketi]# echo -n "123456" | base64
2MTIzNDU2

          创建storage class资源文件

           1
 2[root@k8s-node1 heketi]# vim heketi-sc.yaml
 3apiVersion: v1
 4kind: Secret
 5metadata:
 6  name: heketi-secret
 7  namespace: kube-system
 8type: kubernetes.io/glusterfs
 9data:
10  key: "MTIzNDU2" # heketi认证的密码，经过base64转码
11---
12apiVersion: storage.k8s.io/v1
13kind: StorageClass
14metadata:
15  name: heketi-sc
16  annotations:
17    storageclass.beta.kubernetes.io/is-default-class: "true" # 将 glusterfs 设置为默认存储类型。
18    storageclass.kubesphere.io/supported-access-modes: '["ReadWriteOnce","ReadOnlyMany","ReadWriteMany"]'
19
20parameters:
21
22  clusterid: "c46ab854ec4344b7c25c5d8aeac03beb"   # GlusterFS 集群 ID；可选参数，因为 heketi 可管理多个集群
23  gidMax: "50000"
24  gidMin: "40000"
25  restauthenabled: "true"          # 若heketi开启认证此处也必须开启auth认证
26  resturl: "http://192.168.128.11:8080"
27  restuser: admin
28  secretName: heketi-secret      # name与 heketi-secret.yaml 资源中定义一致
29  secretNamespace: kube-system   # namespace与 heketi-secret.yaml 资源中定义一致
30  volumetype: "replicate:2"
31provisioner: kubernetes.io/glusterfs
32reclaimPolicy: Delete
33volumeBindingMode: Immediate
34allowVolumeExpansion: true

          注意：storageclass资源创建后不可变更，如修改只能删除后重建，所以不能使用 kubectl apply

          应用StorageClass资源到k8s集群：

          1[root@k8s-node1 heketi]# kubectl create -f heketi-sc.yaml
2secret/heketi-secret created
3storageclass.storage.k8s.io/glusterfs created

          查看已经创建的storageclass：

          1[root@k8s-node1 heketi]# kubectl get sc
2NAME                  PROVISIONER               RECLAIMPOLICY   VOLUMEBINDINGMODE   ALLOWVOLUMEEXPANSION   AGE
3heketi-sc (default)   kubernetes.io/glusterfs   Delete          Immediate           true                   37s

          删除storageclass：

          1[root@k8s-node1 ~]# kubectl delete storageclass heketi-sc
2storageclass.storage.k8s.io "heketi-sc" deleted

          （3）创建pvc（heketi-pvc.yaml）

           1[root@k8s-node1 heketi]# vim heketi-pvc.yaml
 2apiVersion: v1
 3kind: PersistentVolumeClaim
 4metadata:
 5  name: heketi-pvc
 6
 7spec:
 8  storageClassName: "heketi-sc" # 如果不设置，会使用默认的 StorageClass
 9  # ReadWriteOnce：简写RWO，读写权限，且只能被单个node挂载；
10  # ReadOnlyMany：简写ROX，只读权限，允许被多个node挂载；
11  # ReadWriteMany：简写RWX，读写权限，允许被多个node挂载；
12  accessModes:
13  - ReadWriteMany
14  resources:
15    requests:
16      storage: 5Gi  # 注意格式，不能写GB

          应用pvc资源

          1[root@glusterfs-node1 heketi]# kubectl create -f heketi-pvc.yaml
2persistentvolumeclaim/heketi-pvc created
