          前提

          • 准备好你的springboot项目jar包

          • 构建vue项目，并打包成tar.gz文件（注意：只要构建后的index.html所在目录的所有文件、文件夹）

          1. ** springboot项目的构建和发布**

          （1）配置Dockerfile

          vim Dockerfile

          内容如下：

           1FROM openjdk:8-jdk-alpine
 2
 3 # 软件源修改为国内清华镜像站
 4RUN sed -i 's/dl-cdn.alpinelinux.org/mirrors.tuna.tsinghua.edu.cn/g' /etc/apk/repositories
 5
 6# 解决时差8小时问题
 7ENV TZ=Asia/Shanghai
 8RUN ln -snf /usr/share/zoneinfo/$TZ /etc/localtime && echo $TZ > /etc/timezone
 9
10# 解决字体问题
11RUN apk add --update ttf-dejavu fontconfig && rm -rf /var/cache/apk/*
12
13ADD ./ruoyi-admin.jar /app.jar
14ENTRYPOINT ["java","-jar","app.jar","--spring.profiles.active=prod"]

          或者：

          1# 已解决时差和字体问题
2FROM registry.cn-hangzhou.aliyuncs.com/conggege325/pub-repository:8-jdk-alpine-timezone-font
3ADD ./ruoyi-admin.jar /app.jar
4ENTRYPOINT ["java","-jar","app.jar","--spring.profiles.active=prod"]

          （2）构建镜像到本地docker中

          1docker build --no-cache -f Dockerfile -t smartbrain:smartbrain-server .

          注意：

          • -t 后面是镜像的名字

          • 最后面有一个点，表示Dockerfile所在目录

          （3）本地测试（可选操作）

          运行：

          1docker run -p 8765:8080 -d smartbrain:smartbrain-server

          然后在宿主机的浏览器中访问：

          http://192.168.0.91:8765

          查看容器日志

          docker logs -f [容器id]

          参数说明：

          -f : 跟踪日志输出

          –since : 显示某个开始时间的所有日志

          -t : 显示时间戳

          –tail : 仅列出最新N条容器日志

          例如：查看从2016年7月1日后的最新10条日志。

          1$ docker logs --since="2016-07-01" --tail=10 378ea24b1bda

          （4）推送至远程仓库

          1[zhangcong@master k8s]$ docker tag 174bf9cb5d15 registry.cn-hangzhou.aliyuncs.com/conggege325/pri-repository:smartbrain-server
2
3[zhangcong@master k8s]$ docker push registry.cn-hangzhou.aliyuncs.com/conggege325/pri-repository:smartbrain-server

          （5）在K8S中发布

          a）创建 namespace
          1[zhangcong@master k8s]$ kubectl create ns smartbrain
          b）创建secret（因为是从私有仓库中拉取）
          1[zhangcong@master k8s]$ kubectl create secret docker-registry ali-secret \
2--docker-server=registry.cn-hangzhou.aliyuncs.com \
3--docker-username='iweiyi10@163.com' \
4--docker-password='zc123456' \
5--namespace=smartbrain

          注意：ali-secret 是创建的secret名 ，username 和password是阿里云仓库的账号和密码

          查看创建的secret

          1[zhangcong@master k8s]$ kubectl get secret -n smartbrain
          c）创建发布的配置文件

          vim smartbrain-server-deploy.yaml

          内容如下：

           1apiVersion: apps/v1
 2kind: Deployment
 3metadata:
 4  name: smartbrain-server-deployment
 5  namespace: smartbrain
 6spec:
 7  replicas: 1
 8  selector:
 9    matchLabels:
10      app: smartbrain-server
11  template:
12    metadata:
13      labels:
14        app: smartbrain-server
15    spec:
16      imagePullSecrets:
17        - name: ali-secret
18      containers:
19        - name: smartbrain-server-container
20          image: registry.cn-hangzhou.aliyuncs.com/conggege325/pri-repository:smartbrain-server
21          ports:
22            - containerPort: 8080
23---
24apiVersion: v1
25kind: Service
26metadata:
27  name: smartbrain-server-service
28  namespace: smartbrain
29spec:
30  ports:
31    - name: http
32      port: 80
33      targetPort: 8080
34  selector:
35    app: smartbrain-server
36  type: NodePort

          应用配置：

          1kubectl apply -f smartbrain-server-deploy.yaml
          1. vue项目的构建和发布

          （0）前提

          准备好nginx.conf

          内容如下：

           1worker_processes  1;
 2
 3events {
 4    worker_connections  1024;
 5}
 6
 7http {
 8    include            mime.types;
 9    default_type       application/octet-stream;
10    sendfile           on;
11    keepalive_timeout  65;
12
13    server {
14        listen       80;
15        server_name  localhost;
16
17        location / {
18            root       /usr/share/nginx/html/;
19            index      index.html  index.htm;
20            try_files  $uri  $uri/  /index.html;
21        }
22
23        location /whoami/ {
24            proxy_set_header Host $http_host;
25            proxy_set_header X-Real-IP $remote_addr;
26            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
27            proxy_pass http://10.99.75.202/;
28        }
29
30        location /prod-api/ {
31            proxy_set_header Host $http_host;
32            proxy_set_header X-Real-IP $remote_addr;
33            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
34            # 此处的ip是springboot部署在k8s中的服务地址
35            proxy_pass http://10.106.166.214/;
36        }
37
38
39        error_page   500 502 503 504  /50x.html;
40        location = /50x.html {
41            root   html;
42        }
43    }
44}

          （1）配置Dockerfile

          vim Dockerfile

          内容如下：

           1#基础镜像
 2FROM docker.io/library/nginx:1.25.3
 3
 4#作者信息
 5MAINTAINER CGG
 6
 7#vue dist包上传到容器nginx默认目录
 8ADD ./dist.tar.gz /usr/share/nginx/html
 9
10#复制本地nginx配置文件覆盖容器nginx默认配置文件
11COPY ./nginx.conf /etc/nginx/nginx.conf
12
13#暴露容器端口
14EXPOSE 80
15
16#指定容器时区
17RUN rm -f /etc/localtime \
18&& ln -sv /usr/share/zoneinfo/Asia/Shanghai /etc/localtime \
19&& echo "Asia/Shanghai" > /etc/timezone
20
21#启动nginx容器服务
22ENTRYPOINT nginx -g "daemon off;"

          （2）构建镜像到本地docker中

          1docker build --no-cache -f Dockerfile -t smartbrain:smartbrain-web .

          注意：

          • -t 后面是镜像的名字

          • 最后面有一个点，表示Dockerfile所在目录

          （3）推送至远程仓库

          1[zhangcong@master web]$ docker tag smartbrain:smartbrain-web  registry.cn-hangzhou.aliyuncs.com/conggege325/pri-repository:smartbrain-web
2
3
4[zhangcong@master web]$ docker push registry.cn-hangzhou.aliyuncs.com/conggege325/pri-repository:smartbrain-web

          （4）在K8S中发布

          a）发布的配置

          vim smartbrain-web-deploy.yaml

          内容如下：

           1apiVersion: apps/v1
 2kind: Deployment
 3metadata:
 4  name: smartbrain-web-depoyment
 5  namespace: smartbrain
 6spec:
 7  replicas: 1
 8  selector:
 9    matchLabels:
10      app: smartbrain-web
11  template:
12    metadata:
13      labels:
14        app: smartbrain-web
15    spec:
16      containers:
17        - name: smartbrain-web-container
18          image: registry.cn-hangzhou.aliyuncs.com/conggege325/pri-repository:smartbrain-web
19          ports:
20            - containerPort: 80
21      imagePullSecrets:
22        - name: ali-secret
23---
24apiVersion: v1
25kind: Service
26metadata:
27  name: smartbrain-web-service
28  namespace: smartbrain
29spec:
30  type: NodePort
31  ports:
32    - port: 80
33      targetPort: 80
34  selector:
35    app: smartbrain-web

          应用配置：

          1kubectl apply -f smartbrain-web-deploy.yaml
          b）网络的配置

          vim smartbrain-ingress.yaml

          内容如下：

           1apiVersion: networking.k8s.io/v1
 2kind: Ingress
 3metadata:
 4  name: smartbrain-ingress
 5  namespace: smartbrain
 6  annotations:
 7    kubernetes.io/ingress.class: nginx
 8spec:
 9  rules:
10  - host: smartbrain.com
11    http:
12      paths:
13        - path: /
14          pathType: Prefix
15          backend:
16            service:
17              name: smartbrain-web-service
18              port:
19                number: 80

          应用配置：

          1kubectl apply -f smartbrain-ingress.yaml
